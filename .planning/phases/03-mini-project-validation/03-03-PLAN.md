         if (artifactPath.startsWith('~/')) {
           return path.join(os.homedir(), artifactPath.slice(2));
         }
         if (artifactPath === '~') {
           return os.homedir();

           return artifactPath;
         }
         return path.join(cwd, artifactPath);
       }
